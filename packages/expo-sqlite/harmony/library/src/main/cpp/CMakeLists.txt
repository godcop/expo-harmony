cmake_minimum_required(VERSION 3.19)
project(expo_sqlite LANGUAGES C CXX)

set(SQLITE_VENDOR "${CMAKE_CURRENT_SOURCE_DIR}/../../../../generated/vendor")
set(SQLITE_PREBUILT "${CMAKE_CURRENT_SOURCE_DIR}/../../../../generated/prebuilt")
set(EXPO_SQLITE_CONFIG "${CMAKE_CURRENT_SOURCE_DIR}/../../../../sqlite-options.json" CACHE FILEPATH "SQLite build options")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${EXPO_SQLITE_CONFIG}")
file(READ "${EXPO_SQLITE_CONFIG}" SQLITE_CONFIG)
string(JSON SQLITE_CONFIG_TYPE TYPE "${SQLITE_CONFIG}")
if(NOT SQLITE_CONFIG_TYPE STREQUAL "OBJECT")
  message(FATAL_ERROR "SQLite build options must be a JSON object")
endif()
function(sqlite_option key default expected_type)
  string(JSON value_type ERROR_VARIABLE missing TYPE "${SQLITE_CONFIG}" "${key}")
  if(missing)
    set(value "${default}")
  else()
    if(NOT value_type STREQUAL expected_type)
      message(FATAL_ERROR "SQLite option ${key} must have type ${expected_type}")
    endif()
    string(JSON value GET "${SQLITE_CONFIG}" "${key}")
  endif()
  set(${key} "${value}" PARENT_SCOPE)
endfunction()
sqlite_option(enableFTS ON BOOLEAN)
sqlite_option(useSQLCipher OFF BOOLEAN)
sqlite_option(useLibSQL OFF BOOLEAN)
sqlite_option(withSQLiteVecExtension OFF BOOLEAN)
sqlite_option(customBuildFlags "" STRING)
string(JSON SQLITE_OPTION_COUNT LENGTH "${SQLITE_CONFIG}")
set(SQLITE_OPTION_NAMES enableFTS useSQLCipher useLibSQL withSQLiteVecExtension customBuildFlags)
if(SQLITE_OPTION_COUNT GREATER 0)
  math(EXPR SQLITE_LAST_OPTION "${SQLITE_OPTION_COUNT} - 1")
  foreach(index RANGE ${SQLITE_LAST_OPTION})
    string(JSON option_name MEMBER "${SQLITE_CONFIG}" ${index})
    if(NOT option_name IN_LIST SQLITE_OPTION_NAMES)
      message(FATAL_ERROR "Unknown SQLite build option: ${option_name}")
    endif()
  endforeach()
endif()
if(useLibSQL AND useSQLCipher)
  message(FATAL_ERROR "useLibSQL and useSQLCipher select different database backends and cannot be combined")
endif()
if(useLibSQL AND withSQLiteVecExtension)
  message(FATAL_ERROR "Expo's libSQL backend does not support loadExtension; sqlite-vec requires SQLite or SQLCipher")
endif()

add_library(expo_sqlite SHARED SQLiteBinding.cpp)
target_compile_features(expo_sqlite PRIVATE cxx_std_17)
target_compile_definitions(expo_sqlite PRIVATE
  EXPO_SQLITE_USE_LIBSQL=$<BOOL:${useLibSQL}>
  EXPO_SQLITE_WITH_VEC=$<BOOL:${withSQLiteVecExtension}>)
set_target_properties(expo_sqlite PROPERTIES
  C_VISIBILITY_PRESET hidden CXX_VISIBILITY_PRESET hidden
  VISIBILITY_INLINES_HIDDEN YES)
target_link_libraries(expo_sqlite PRIVATE libace_napi.z.so libdl.so libm.so)
target_link_options(expo_sqlite PRIVATE -Wl,--exclude-libs,ALL)

if(useLibSQL)
  set(LIBSQL_LIBRARY "${SQLITE_PREBUILT}/${OHOS_ARCH}/libsql_experimental.a")
  if(NOT EXISTS "${LIBSQL_LIBRARY}")
    message(FATAL_ERROR "Missing ${LIBSQL_LIBRARY}; reinstall the complete @expo-harmony/expo-sqlite npm package")
  endif()
  add_library(expo_libsql STATIC IMPORTED)
  set_target_properties(expo_libsql PROPERTIES IMPORTED_LOCATION "${LIBSQL_LIBRARY}")
  target_link_libraries(expo_sqlite PRIVATE expo_libsql)
  target_include_directories(expo_sqlite PRIVATE "${SQLITE_VENDOR}/libsql")
  message(STATUS "libSQL uses its own upstream build flags; enableFTS/customBuildFlags apply only to SQLite/SQLCipher")
else()
  if(useSQLCipher)
    set(SQLITE_SOURCE "${SQLITE_VENDOR}/sqlcipher")
    set(CRYPTO_ROOT "${SQLITE_PREBUILT}/${OHOS_ARCH}/openssl")
    if(NOT EXISTS "${CRYPTO_ROOT}/lib/libcrypto.a")
      message(FATAL_ERROR "Missing ${CRYPTO_ROOT}/lib/libcrypto.a; reinstall the complete @expo-harmony/expo-sqlite npm package")
    endif()
    add_library(expo_sqlite_crypto STATIC IMPORTED)
    set_target_properties(expo_sqlite_crypto PROPERTIES IMPORTED_LOCATION "${CRYPTO_ROOT}/lib/libcrypto.a")
    target_include_directories(expo_sqlite PRIVATE "${CRYPTO_ROOT}/include")
    target_link_libraries(expo_sqlite PRIVATE expo_sqlite_crypto)
    target_compile_definitions(expo_sqlite PRIVATE SQLITE_HAS_CODEC=1 SQLCIPHER_CRYPTO_OPENSSL
      SQLITE_EXTRA_INIT=sqlcipher_extra_init SQLITE_EXTRA_SHUTDOWN=sqlcipher_extra_shutdown NDEBUG)
  else()
    set(SQLITE_SOURCE "${SQLITE_VENDOR}")
  endif()
  target_sources(expo_sqlite PRIVATE "${SQLITE_SOURCE}/sqlite3.c")
  target_include_directories(expo_sqlite PRIVATE "${SQLITE_SOURCE}")
  target_compile_definitions(expo_sqlite PRIVATE
    SQLITE_ENABLE_BYTECODE_VTAB=1 SQLITE_TEMP_STORE=2
    SQLITE_ENABLE_SESSION=1 SQLITE_ENABLE_PREUPDATE_HOOK=1 SQLITE_ENABLE_MATH_FUNCTIONS=1)
  if(enableFTS)
    target_compile_definitions(expo_sqlite PRIVATE SQLITE_ENABLE_FTS4=1 SQLITE_ENABLE_FTS3_PARENTHESIS=1 SQLITE_ENABLE_FTS5=1)
  endif()
  separate_arguments(SQLITE_CUSTOM_FLAGS UNIX_COMMAND "${customBuildFlags}")
  target_compile_options(expo_sqlite PRIVATE ${SQLITE_CUSTOM_FLAGS})
endif()

if(withSQLiteVecExtension)
  add_library(expo_sqlite_vec SHARED "${SQLITE_VENDOR}/sqlite-vec/sqlite-vec.c")
  target_include_directories(expo_sqlite_vec PRIVATE "${SQLITE_VENDOR}/sqlite-vec")
  # sqlite-vec 0.1.7 assumes BSD integer typedefs; expose the standard musl header.
  target_compile_options(expo_sqlite_vec PRIVATE -include sys/types.h)
  if(OHOS_ARCH STREQUAL "arm64-v8a")
    target_compile_definitions(expo_sqlite_vec PRIVATE SQLITE_VEC_ENABLE_NEON)
  endif()
  target_link_libraries(expo_sqlite_vec PRIVATE libm.so)
endif()
